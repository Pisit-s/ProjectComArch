import { readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { ImageResponse } from 'next/og';

// The navbar logo chip (navy to blue gradient, white "CoE" in Chakra Petch) as a PNG app icon.
export async function renderLogoIcon(size: number, { rounded }: { rounded: boolean }) {
  const font = await readFile(
    join(
      process.cwd(),
      'node_modules/@fontsource/chakra-petch/files/chakra-petch-latin-700-normal.woff',
    ),
  );
  return new ImageResponse(
    <div
      style={{
        width: '100%',
        height: '100%',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        background: 'linear-gradient(135deg, #102880 0%, #2563eb 100%)',
        borderRadius: rounded ? size * 0.22 : 0,
        color: '#ffffff',
        fontFamily: 'Chakra Petch',
        fontWeight: 700,
        fontSize: size * 0.42,
        letterSpacing: -size * 0.01,
      }}
    >
      CoE
    </div>,
    {
      width: size,
      height: size,
      fonts: [{ name: 'Chakra Petch', data: font, weight: 700, style: 'normal' }],
    },
  );
}
