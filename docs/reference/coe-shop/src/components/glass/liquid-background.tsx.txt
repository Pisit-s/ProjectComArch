import type { CSSProperties } from 'react';

// Each blob is a plain radial gradient sized to look like the old `blur(140px)` disc. Same soft field,
// but nothing for the browser to re-blur: big CSS blur filters were a main reason iPhone Safari lagged.
const blob = (color: string, style: CSSProperties) => ({
  ...style,
  background: `radial-gradient(closest-side, ${color} 0%, ${color} 30%, transparent 100%)`,
});

const BLOBS: CSSProperties[] = [
  blob('rgb(16 40 128 / 0.12)', {
    width: '55.5rem',
    height: '55.5rem',
    top: '-18.75rem',
    left: '-16.75rem',
  }),
  blob('rgb(37 99 235 / 0.16)', {
    width: '47.5rem',
    height: '47.5rem',
    top: 'calc(33% - 8.75rem)',
    left: '-22.75rem',
  }),
  blob('rgb(56 182 255 / 0.25)', {
    width: '49.5rem',
    height: '49.5rem',
    top: '-14.75rem',
    right: '-18.75rem',
  }),
  blob('rgb(56 182 255 / 0.22)', {
    width: '52rem',
    height: '52rem',
    bottom: '-21.5rem',
    left: 'calc(25% - 9.5rem)',
  }),
  blob('rgb(34 211 238 / 0.12)', {
    width: '41.5rem',
    height: '41.5rem',
    bottom: '-16.75rem',
    right: 'calc(25% - 8.75rem)',
  }),
];

/**
 * The colour field every glass surface sits on: a light blue gradient, five soft colour blobs and a faint
 * blueprint grid. Same recipe as ThodBaan, in CoE blue. A fixed layer behind the page; static on phones.
 */
export function LiquidBackground() {
  return (
    <div
      aria-hidden
      className="pointer-events-none fixed inset-0 -z-10 overflow-hidden bg-[linear-gradient(180deg,#f5faff_0%,#ecf5ff_55%,#ffffff_100%)]"
    >
      <div className="absolute inset-0 bg-gradient-to-br from-navy-900/8 via-transparent to-sky-400/8" />
      {BLOBS.map((style, i) => (
        <div key={i} className="absolute rounded-full" style={style} />
      ))}
      <div className="tb-blueprint absolute -inset-8 opacity-25 mask-[radial-gradient(120%_100%_at_50%_0%,black,transparent_80%)]" />
    </div>
  );
}
