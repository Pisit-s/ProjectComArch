import type { Metadata, Viewport } from 'next';
import {
  Barlow_Condensed,
  Chakra_Petch,
  IBM_Plex_Sans_Thai,
  JetBrains_Mono,
} from 'next/font/google';
import { NextIntlClientProvider } from 'next-intl';
import { getLocale, getTranslations } from 'next-intl/server';
import { GlassPointer } from '@/components/glass/glass-pointer';
import { LiquidBackground } from '@/components/glass/liquid-background';
import { InAppBrowserGate } from '@/components/shop/in-app-browser-gate';
import { CartHydrator } from '@/components/providers/cart-hydrator';
import { MotionProvider } from '@/components/providers/motion-provider';
import { Toaster } from '@/components/ui/sonner';
import { siteUrl } from '@/lib/env';
import './globals.css';

const chakraPetch = Chakra_Petch({
  variable: '--font-chakra-petch',
  weight: ['400', '500', '600', '700'],
  subsets: ['thai', 'latin'],
  display: 'swap',
});

const plexSansThai = IBM_Plex_Sans_Thai({
  variable: '--font-plex-sans-thai',
  weight: ['400', '500', '600', '700'],
  subsets: ['thai', 'latin'],
  display: 'swap',
});

const jetbrainsMono = JetBrains_Mono({
  variable: '--font-jetbrains-mono',
  weight: ['400', '500', '700'],
  subsets: ['latin'],
  display: 'swap',
});

// only for the T-shirt name preview; not preloaded on every page
const barlowCondensed = Barlow_Condensed({
  variable: '--font-barlow-condensed',
  weight: ['500'],
  subsets: ['latin'],
  display: 'swap',
  preload: false,
});

export async function generateMetadata(): Promise<Metadata> {
  const t = await getTranslations('meta');
  return {
    metadataBase: new URL(siteUrl()),
    title: { default: t('title'), template: `%s · ${t('title')}` },
    description: t('description'),
  };
}

export const viewport: Viewport = {
  width: 'device-width',
  initialScale: 1,
  viewportFit: 'cover',
  colorScheme: 'light',
  themeColor: '#ECF5FF',
};

export default async function RootLayout({ children }: LayoutProps<'/'>) {
  const locale = await getLocale();
  return (
    <html
      lang={locale}
      className={`${chakraPetch.variable} ${plexSansThai.variable} ${jetbrainsMono.variable} ${barlowCondensed.variable} h-full antialiased`}
    >
      <body className="min-h-full">
        <NextIntlClientProvider>
          <MotionProvider>
            <LiquidBackground />
            <GlassPointer />
            <CartHydrator />
            <InAppBrowserGate />
            {children}
            <Toaster />
          </MotionProvider>
        </NextIntlClientProvider>
      </body>
    </html>
  );
}
