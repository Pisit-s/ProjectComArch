/*
 * Touch screens: no backdrop blur and no pointer-following light. iPhone Safari re-blurs every glass
 * surface on every scroll frame (the page scrolled at ~27fps even on a Mac's WebKit); over this already
 * soft background the blur was barely visible anyway. Panels that float over content get a more solid
 * fill instead (see the navbar and sticky cart bar).
 */
@media (hover: none) {
  .glass,
  .glass-raised {
    -webkit-backdrop-filter: none;
    backdrop-filter: none;
  }
  .glass-interactive::before,
  .glass-interactive::after {
    content: none;
  }
}

@supports not ((backdrop-filter: blur(1px)) or (-webkit-backdrop-filter: blur(1px))) {
  .glass,
  .glass-raised {
    background: rgb(255 255 255 / 0.92);
  }
}

@media (prefers-reduced-transparency: reduce) {
  .glass,
  .glass-raised {
    background: rgb(255 255 255 / 0.95);
    -webkit-backdrop-filter: none;
    backdrop-filter: none;
  }
}

@media (prefers-reduced-motion: reduce) {
  .tb-blueprint,
  .tb-rise,
  .tb-pop,
  .tb-glow,
  .cursor-blink {
    animation: none;
  }
  .glass-interactive,
  .glass-interactive::after,
  .glass-interactive::before {
    transition: none;
  }
  .glass-interactive:hover,
  .glass-interactive:active {
    transform: none;
  }
}
